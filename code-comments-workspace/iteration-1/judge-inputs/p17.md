```
   1  """Cache em memória com expiração por tempo (TTL).
   2  
   3  Cada valor guardado vale por `ttl_s` segundos; depois disso é tratado como
   4  inexistente. Não é thread-safe e não é compartilhado entre processos.
   5  """
   6  
   7  import time
   8  
   9  # Limite de entradas no cache. Ao atingir esse limite, inserir uma chave nova
  10  # remove a entrada mais antiga para abrir espaço.
  11  MAX_ENTRIES = 1024
  12  
  13  
  14  class TTLCache:
  15      def __init__(self, ttl_s, clock=time.monotonic):
  16          # ttl_s: tempo de vida de cada entrada, em segundos.
  17          # clock: função que devolve o "agora" em segundos. O padrão é
  18          # time.monotonic (não é afetado por mudanças no relógio do sistema).
  19          # Existe como parâmetro para os testes poderem simular a passagem do
  20          # tempo sem usar sleep.
  21          self._ttl_s = ttl_s
  22          self._clock = clock
  23          # chave -> (valor, instante em que foi gravado)
  24          self._entries = {}
  25  
  26      def get(self, key, default=None):
  27          """Devolve o valor de `key`, ou `default` se não existir ou tiver expirado."""
  28          entry = self._entries.get(key)
  29          if entry is None:
  30              return default
  31          value, stored_at = entry
  32          # A expiração é preguiçosa: entradas vencidas só são removidas quando
  33          # alguém tenta lê-las. Até lá, continuam ocupando espaço no cache.
  34          # Com idade exatamente igual ao TTL, a entrada ainda é válida (> e não >=).
  35          if self._clock() - stored_at > self._ttl_s:
  36              del self._entries[key]
  37              return default
  38          return value
  39  
  40      def put(self, key, value):
  41          """Grava `value` em `key`, reiniciando a contagem do TTL dessa chave."""
  42          # Só despeja quando a chave é nova: sobrescrever uma chave existente
  43          # não aumenta o tamanho do cache.
  44          if key not in self._entries and len(self._entries) >= MAX_ENTRIES:
  45              # Despeja a entrada gravada há mais tempo (varredura linear, O(n)).
  46              # Pode ser uma entrada ainda válida; não há prioridade por uso.
  47              oldest = min(self._entries, key=lambda k: self._entries[k][1])
  48              del self._entries[oldest]
  49          self._entries[key] = (value, self._clock())
```
