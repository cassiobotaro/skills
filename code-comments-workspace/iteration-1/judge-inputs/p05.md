   2  
   3  # Limite de entradas; ao atingi-lo, put() descarta a entrada mais antiga.
   4  MAX_ENTRIES = 1024
   5  
   6  
   7  # Cache em memória em que cada entrada expira `ttl_s` segundos depois de gravada.
   8  # A expiração é preguiçosa: uma entrada vencida só sai do dicionário quando é lida
   9  # com get() ou quando é expulsa por falta de espaço. Até lá ela continua ocupando
  10  # uma das MAX_ENTRIES vagas. Não há lock; não use a mesma instância em várias threads
  11  # sem sincronização externa.
  12  class TTLCache:
  13      # `clock` é uma função sem argumentos que devolve segundos. O padrão,
  14      # time.monotonic, não é afetado por ajustes do relógio do sistema.
  15      def __init__(self, ttl_s, clock=time.monotonic):
  16          self._ttl_s = ttl_s
  17          self._clock = clock
  18          # chave -> (valor, instante em que foi gravada)
  19          self._entries = {}
  20  
  21      # Devolve `default` se a chave não existe ou já expirou (e nesse caso a remove).
  22      # Uma entrada com idade exatamente igual a ttl_s ainda é válida.
  23      # Ler não renova o prazo: o TTL conta a partir do put().
  24      def get(self, key, default=None):
  25          entry = self._entries.get(key)
  26          if entry is None:
  27              return default
  28          value, stored_at = entry
  29          if self._clock() - stored_at > self._ttl_s:
  30              del self._entries[key]
  31              return default
  32          return value
  33  
  34      # Regravar uma chave existente reinicia o prazo dela e nunca expulsa ninguém.
  35      def put(self, key, value):
  36          if key not in self._entries and len(self._entries) >= MAX_ENTRIES:
  37              # Cache cheio e chave nova: expulsa a entrada gravada há mais tempo
  38              # (pode ser uma já expirada ou não). É uma busca linear, O(n) por put()
  39              # com o cache cheio.
  40              oldest = min(self._entries, key=lambda k: self._entries[k][1])
  41              del self._entries[oldest]
