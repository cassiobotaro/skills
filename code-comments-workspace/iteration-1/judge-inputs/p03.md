   2  
   3  # Limite de itens no cache. Ao atingir esse número, inserir uma chave nova
   4  # descarta a entrada mais antiga (ver TTLCache.put).
   5  MAX_ENTRIES = 1024
   6  
   7  
   8  class TTLCache:
   9      """Cache em memória em que cada entrada expira após `ttl_s` segundos.
  10  
  11      A expiração é "preguiçosa": entradas vencidas só são removidas quando
  12      alguém chama `get` para a mesma chave (ou quando são despejadas por
  13      falta de espaço em `put`). Não há thread de limpeza em segundo plano.
  14  
  15      Não é thread-safe: use um lock externo se for compartilhar entre threads.
  16      """
  17  
  18      def __init__(self, ttl_s, clock=time.monotonic):
  19          # Tempo de vida de cada entrada, em segundos.
  20          self._ttl_s = ttl_s
  21          # Função que devolve "agora" em segundos. É injetável para os testes
  22          # poderem avançar o tempo sem dar sleep. O padrão, time.monotonic, não
  23          # é afetado por mudanças no relógio do sistema.
  24          self._clock = clock
  25          # chave -> (valor, instante em que foi gravado)
  26          self._entries = {}
  27  
  28      def get(self, key, default=None):
  29          """Devolve o valor de `key`, ou `default` se não existir ou tiver expirado."""
  30          entry = self._entries.get(key)
  31          if entry is None:
  32              return default
  33          value, stored_at = entry
  34          # Só expira quando a idade passa do TTL (estritamente maior):
  35          # com idade exatamente igual a ttl_s a entrada ainda é válida.
  36          if self._clock() - stored_at > self._ttl_s:
  37              # Remove na leitura, já que nada mais limpa entradas vencidas.
  38              del self._entries[key]
  39              return default
  40          return value
  41  
  42      def put(self, key, value):
  43          """Grava `value` em `key`, reiniciando o TTL dessa chave."""
  44          # Só despeja se a chave é nova e o cache está cheio; sobrescrever uma
  45          # chave existente não aumenta o tamanho.
  46          if key not in self._entries and len(self._entries) >= MAX_ENTRIES:
  47              # Despeja a entrada gravada há mais tempo (não a menos usada: ler
  48              # não atualiza `stored_at`). É uma busca linear, O(n), aceitável
  49              # com MAX_ENTRIES pequeno. Pode despejar uma entrada ainda válida
  50              # mesmo havendo vencidas, pois não procura por elas.
  51              oldest = min(self._entries, key=lambda k: self._entries[k][1])
  52              del self._entries[oldest]
  53          self._entries[key] = (value, self._clock())
