```
   1  import re
   2  import unicodedata
   3  
   4  
   5  def slugify(title: str) -> str:
   6      """Lowercase ASCII slug of `title`; anything that is not a letter or digit becomes '-'.
   7  
   8      slugify("Olá, Mundo!") -> "ola-mundo"
   9      """
  10      ascii_title = unicodedata.normalize("NFKD", title).encode("ascii", "ignore").decode()
  11      return re.sub(r"[^a-z0-9]+", "-", ascii_title.lower()).strip("-")
  12  
  13  
  14  def unique_slug(title: str, existing: set[str]) -> str:
  15      """Slug of `title` that is not in `existing`; `existing` is not modified.
  16  
  17      Taken slugs get a numeric suffix starting at 2: with {"a", "a-2"}, "A" -> "a-3".
  18      A title with no letters or digits has slug "", which is treated like any other.
  19      """
  20      base = slugify(title)
  21      slug = base
  22      n = 2
  23      while slug in existing:
  24          slug = f"{base}-{n}"
  25          n += 1
  26      return slug
```
