### slugs/slugify.py (CHANGED by the run)

```
   1  import re
   2  import unicodedata
   3  
   4  
   5  def slugify(title: str) -> str:
   6      """Lowercase ASCII slug of `title`; anything that is not a letter or digit becomes '-'.
   7  
   8      slugify("Olá, Mundo!") -> "ola-mundo"
   9      """
  10      ascii_title = unicodedata.normalize("NFKD", title).encode("ascii", "ignore").decode()
  11      return re.sub(r"[^a-z0-9]+", "-", ascii_title.lower()).strip("-")
  12  
  13  
  14  def unique_slug(title: str, existing: set[str]) -> str:
  15      """Slug of `title` that is not in `existing`, suffixed -2, -3, ... if taken.
  16  
  17      unique_slug("Olá, Mundo!", {"ola-mundo"}) -> "ola-mundo-2"
  18      """
  19      base = slugify(title)
  20      slug = base
  21      n = 2
  22      while slug in existing:
  23          slug = f"{base}-{n}"
  24          n += 1
  25      return slug
```
