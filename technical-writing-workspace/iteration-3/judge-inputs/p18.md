### docs/pipeline.md

```
   1  # Export pipeline
   2  
   3  This page describes what happens to an export request.
   4  
   5  ## Request
   6  
   7  When an export is requested, the payload is validated. Requests above 100,000 rows are
   8  rejected. If the payload is valid, a job is enqueued and a `202` is returned.
   9  
  10  ## Processing
  11  
  12  The job is picked up and the CSV file is generated. The file is encrypted before it is
  13  uploaded to the bucket. If something goes wrong, the job is retried up to three times,
  14  after which it is moved to the dead-letter queue.
  15  
  16  ## Delivery
  17  
  18  Once the upload is done, a download link is sent by email. The link is expired after
  19  24 hours.
  20  
  21  ## Backups
  22  
  23  Backups of the bucket are rotated weekly.
```

### exporter/worker.py

```
   1  import json
   2  import os
   3  
   4  import boto3
   5  import pika
   6  from cryptography.fernet import Fernet
   7  
   8  from exporter.notifier import notify
   9  
  10  MAX_ATTEMPTS = 3
  11  
  12  
  13  def build_csv(job: dict) -> bytes:
  14      rows = [f"{job['account_id']},{n}" for n in range(job.get("rows", 0))]
  15      return "\n".join(rows).encode()
  16  
  17  
  18  def handle(channel, method, properties, body) -> None:
  19      job = json.loads(body)
  20      attempts = (properties.headers or {}).get("x-attempts", 0) + 1
  21      try:
  22          data = Fernet(os.environ["EXPORT_KEY"]).encrypt(build_csv(job))
  23          key = f"{job['account_id']}/{method.delivery_tag}.csv.enc"
  24          boto3.client("s3").put_object(Bucket=os.environ["EXPORT_BUCKET"], Key=key, Body=data)
  25          notify(job["account_id"], key)
  26      except Exception:
  27          target = "exports.dead" if attempts >= MAX_ATTEMPTS else "exports"
  28          channel.basic_publish(
  29              exchange="",
  30              routing_key=target,
  31              body=body,
  32              properties=pika.BasicProperties(headers={"x-attempts": attempts}),
  33          )
  34      channel.basic_ack(method.delivery_tag)
  35  
  36  
  37  def main() -> None:
  38      connection = pika.BlockingConnection(pika.URLParameters(os.environ["QUEUE_URL"]))
  39      channel = connection.channel()
  40      channel.basic_consume(queue="exports", on_message_callback=handle)
  41      channel.start_consuming()
  42  
  43  
  44  if __name__ == "__main__":
  45      main()
```

```
   1  """Move messages from exports.dead back to exports.
   2  
   3      uv run python scripts/requeue_dead.py --dry-run    count the messages, move nothing
   4      uv run python scripts/requeue_dead.py --limit 50   move at most 50 messages
   5  """
   6  import argparse
   7  import os
   8  
   9  import pika
  10  
  11  
  12  def main() -> None:
  13      parser = argparse.ArgumentParser()
  14      parser.add_argument("--dry-run", action="store_true")
  15      parser.add_argument("--limit", type=int, default=100)
  16      args = parser.parse_args()
  17  
  18      connection = pika.BlockingConnection(pika.URLParameters(os.environ["QUEUE_URL"]))
  19      channel = connection.channel()
  20      moved = 0
  21      while moved < args.limit:
  22          method, properties, body = channel.basic_get("exports.dead")
  23          if method is None:
  24              break
  25          if args.dry_run:
  26              channel.basic_nack(method.delivery_tag, requeue=True)
  27          else:
  28              channel.basic_publish(exchange="", routing_key="exports", body=body)
  29              channel.basic_ack(method.delivery_tag)
  30          moved += 1
  31      print(f"{'would move' if args.dry_run else 'moved'} {moved} message(s)")
  32  
  33  
  34  if __name__ == "__main__":
  35      main()
```
